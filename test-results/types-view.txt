7 | /** A film or show in the catalog. */
8 | export interface ContentItem {

10 |   id: string

12 |   description: string
13 |   /** One-line tagline, shown in the hero section. */
14 |   logline?: string
15 |   /** Portrait (2:3) card image. */
16 |   thumbnailUrl: string
17 |   /** Landscape backdrop used for the hero and wide cards. */
18 |   backdropUrl: string
19 |   /** Release year. */
20 |   year: number

22 |   rating: string

25 |   durationMinutes: number

27 |   stars?: number

29 |   progressPercent?: number
30 | }
